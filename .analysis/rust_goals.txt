### crates/horizon-client/src/mobs/aquatic.rs :: FollowFlockLeader
 -- can_use
{
        if m.extra.followers > 0 {
            return false;
        }
        if follows(cx.w, m.extra.leader) {
            return true;
        }
        if self.next_start > 0 {
            self.next_start -= 1;
            return false;
        }
        self.next_start = Self::next_start_tick(m);
        // The fish of its kind around it that lead a school with room or
        // follow no one, in level order; itself as it is now.
        let species = m.species();
        let area = m.bounds().inflate(Vec3::new(8.0, 8.0, 8.0));
        let mut near: Vec<&crate::mobs::Seen> =
            cx.w.others_near(area)
                .filter(|s| s.species == species && s.alive && s.bounds.intersects(area))
                .collect();
        // `getEntitiesOfClass`: the sections by x, z, y; in each, as they entered it.
        near.sort_by_key(|s| (s.section.0[0], s.section.0[2], s.section.0[1], s.section.1));
        let candidates: Vec<(u32, Option<u32>, i32)> = near
            .into_iter()
            .map(|s| {
                if s.id == m.id {
                    (s.id, m.extra.leader, m.extra.followers)
                } else {
                    (s.id, s.school.0, s.school.1)
                }
            })
            .filter(|&(_, leader, followers)| {
                can_be_followed(species, followers) || !follows(cx.w, leader)
            })
            .collect();
        let (leader, school_size) = candidates
            .iter()
            .find(|c| can_be_followed(species, c.2))
            .map_or((m.id, m.extra.followers + 1), |c| (c.0, c.2 + 1));
        // addFollowers: the limit counts the leader before it is skipped.
        let room = (max_school_size(species) - school_size).max(0) as usize;
        for &(id, ..) in candidates
            .iter()
            .filter(|c| !follows(cx.w, c.1))
            .take(room)
            .filter(|c| c.0 != leader)
        {
            if id == m.id {
                m.extra.leader = Some(leader);
                cx.events.push(MobEvent::SchoolSize { leader, by: 1 });
            } else {
                cx.events
                    .push(MobEvent::StartFollowing { fish: id, leader });
            }
        }
        follows(cx.w, m.extra.leader)
    }

 -- can_continue
{
        m.extra
            .leader
            .and_then(|l| cx.w.seen(l))
            .is_some_and(|l| l.alive && m.distance_sqr(l.position) <= 121.0)
    }

### crates/horizon-client/src/mobs/caravan.rs :: FollowCaravan
 -- can_use
{
        if m.extra.leash.is_some() || in_caravan(m) {
            return false;
        }
        let area = m.bounds().inflate(Vec3::new(9.0, 4.0, 9.0));
        let mine = m.position;
        let id = m.id;
        let candidates: Vec<&Seen> = cx
            .w
            .others
            .iter()
            .filter(|s| s.id != id && is_llama(s.species) && s.alive && s.bounds.intersects(area))
            .collect();
        let mut closest: Option<&Seen> = None;
        let mut best = f64::MAX;
        for c in &candidates {
            if c.caravan != 0 && !has_tail(cx, c) {
                let d = mine.distance_squared(c.position);
                if d <= best {
                    best = d;
                    closest = Some(c);
                }
            }
        }
        if closest.is_none() {
            for c in &candidates {
                if is_leashed_seen(c) && !has_tail(cx, c) {
                    let d = mine.distance_squared(c.position);
                    if d <= best {
                        best = d;
                        closest = Some(c);
                    }
                }
            }
        }
        let Some(head) = closest else {
            return false;
        };
        if best < 4.0 {
            return false;
        }
        if !is_leashed_seen(head) && !first_is_leashed(cx, head.caravan, false, 1) {
            return false;
        }
        m.extra.ints[1] = head.id as i32 + 1;
        true
    }

 -- can_continue
{
        let caravan = head_of(m);
        if caravan == 0 {
            return false;
        }
        let Some(head) = cx.w.seen(caravan - 1).filter(|h| h.alive) else {
            return false;
        };
        if !first_is_leashed(cx, caravan, false, 0) {
            return false;
        }
        // Two llamas that joined the same one in the same tick: the one with
        // the higher id leaves and looks for another place.
        if cx
            .w
            .others
            .iter()
            .any(|o| o.id < m.id && is_llama(o.species) && o.alive && o.caravan == caravan)
        {
            return false;
        }
        let dist_sqr = m.position.distance_squared(head.position);
        if dist_sqr > 676.0 {
            if self.speed <= 3.0 {
                self.speed *= 1.2;
                self.dist_check = reduced(40);
                return true;
            }
            if self.dist_check == 0 {
                return false;
            }
        }
        if self.dist_check > 0 {
            self.dist_check -= 1;
        }
        true
    }

{
        if m.random.next_int_bound(self.interval).unwrap_or(1) != 0 {
            return false;

        let [sx, sz] = motion_step(m.yaw);
        let (bx, by, bz) = (
            m.position.x.floor() as i32,
            m.position.y.floor() as i32,

        );
        let w = cx.w;
        [0, 1, 4, 5, 6, 7].into_iter().all(|i| {

                && !w.has_tag("minecraft:blocks_dolphin_jump", w.block_name(x, by, z))
                && w.facts.is_air(w.world.block(x, by + 1, z))
                && w.facts.is_air(w.world.block(x, by + 2, z))
        })
    }

        (yd * yd >= f64::from(0.03f32)
            || m.head_pitch == 0.0
            || m.head_pitch.abs() >= 10.0
            || !m.fluids.in_water)
            && !m.on_ground

### crates/horizon-client/src/mobs/frog.rs :: LongJump
 -- can_use
{
        // The activity: no cooldown, and for a frog out of water (a frog in
        // water is in its swim activity; a goat has none, so its check below
        // sets the cooldown instead). FrogAi.initJumpActivity also wants
        // TEMPTING_PLAYER and BREED_TARGET absent, and the LAY_SPAWN activity
        // (IS_PREGNANT) goes before LONG_JUMP. The tempting player and the
        // breed target are not gated: Tempt and Breed hold MOVE at a better priority.
        if jump_cooldown(m).is_some()
            || m.species() == Species::Frog && is_pregnant(m)
            || m.species() != Species::Goat && m.fluids.in_water
        {
            return false;
        }
        // GoatAi.initLongJumpActivity's other memories, and the activity
        // order of GoatAi.updateActivity.
        if m.species() == Species::Goat && !super::goat::long_jump_activity(m, cx.w) {
            return false;
        }
        // checkExtraStartConditions: on the ground, not in water or lava or
        // on honey.
        let (x, y, z) = (
            m.position.x.floor() as i32,
            m.position.y.floor() as i32,
            m.position.z.floor() as i32,
        );
        let can = m.on_ground
            && !m.fluids.in_water
            && !m.fluids.in_lava
            && cx.w.block_name(x, y, z) != "minecraft:honey_block";
        if !can {
            sample_cooldown(m, cx.lr, 2);
        }
        can
    }

 -- can_continue
{
        if self.mid_jump {
            // LongJumpMidJump starts the tick after the leap and runs until
            // the mob is on the ground again, for 100 ticks at most; water
            // switches a frog's brain to the swim activity, which stops it
            // too (a goat has none).
            match self.in_air {
                None => {
                    self.in_air = Some(0);
                    set_discard_friction(m, true);
                    set_jump_pose(m, cx.w, sync::LONG_JUMPING);
                }
                Some(t) => self.in_air = Some(t + 1),
            }
            return !m.on_ground
                && (m.species() == Species::Goat || !m.fluids.in_water)
                && self.in_air.is_some_and(|t| t <= MID_JUMP_DURATION);
        }
        let valid = self.initial == Some(m.position)
            && self.tries > 0
            && !m.fluids.in_water
            && !(self.chosen.is_none() && self.candidates.is_empty());
        if !valid {
            sample_cooldown(m, cx.lr, 2);
        }
        valid
    }

            }) || m.fire_ticks > 0 && !m.kind.flags.fire_immune

### crates/horizon-client/src/mobs/goals.rs :: LookAtPlayer
 -- can_use
{
        if m.random.next_float() >= self.probability {
            return false;
        }
        self.look_at = None;
        let range = f64::from(self.range);
        if !self.mobs {
            // level.getNearestPlayer(forNonCombat().range(lookDistance)).
            if m.can_see_player(cx.w, range) {
                self.look_at = Some(Who::Player);
            }
            return self.look_at.is_some();
        }
        // level.getNearestEntity(mobs in the box inflated by (range, 3, range)).
        let area = m.bounds().inflate(Vec3::new(range, 3.0, range));
        let (id, only) = (m.id, self.only);
        let at = Vec3::new(m.position.x, m.eye_y(), m.position.z);
        let candidates: Vec<(u32, Vec3)> =
            cx.w.others_near(area)
                .filter(|s| s.id != id && s.alive && s.bounds.intersects(area))
                .filter(|s| only.is_none_or(|o| s.species == o))
                .map(|s| (s.id, s.position))
                .collect();
        let mut best: Option<(u32, f64)> = None;
        for (cid, pos) in candidates {
            let reach = range.max(2.0);
            if m.distance_sqr(pos) > reach * reach || !m.sees(cx.w, Who::Mob(cid)) {
                continue;
            }
            let d = pos.distance_squared(at);
            if best.is_none_or(|(_, b)| d < b) {
                best = Some((cid, d));
            }
        }
        self.look_at = best.map(|(cid, _)| Who::Mob(cid));
        self.look_at.is_some()
    }

{
        let Some((pos, _)) = self.look_target(cx.w) else {
            return false;
        };
        m.distance_sqr(pos) <= f64::from(self.range * self.range) && self.look_time > 0
    }

### crates/horizon-client/src/mobs/goals.rs :: NearestTarget
 -- can_use
{
        if let Some(f) = self.gate
            && !f(m, cx.w)
        {
            return false;
        }
        if self.interval > 0 && m.random.next_int_bound(self.interval).unwrap_or(1) != 0 {
            return false;

        if let Some(f) = self.when
            && !f(m, cx.w)

        self.found = None;
        // NearestAttackableTargetGoal.findTarget: TargetingConditions.forCombat()
        // always checks the line of sight; mustSee and mustReach only matter
        // to TargetGoal.canContinueToUse and canAttack.
        match &self.prey {
            Prey::Player => {
                if m.can_target_player(cx.w, range, true) {
                    self.found = Some(Who::Player);
                }
            }
            prey => {
                let pos = Vec3::new(m.position.x, m.eye_y(), m.position.z);
                let id = m.id;
                let attacker = m.species();
                let candidates: Vec<(u32, Vec3)> =
                    cx.w.nearby(pos, range, move |s| {
                        s.id != id
                            && !s.playing_dead
                            && !illager_spares(attacker, s)
                            && match prey {
                                Prey::Species(list) => list.contains(&s.species),
                                Prey::Filter(f) => f(s),
                                Prey::Player => false,
                            }
                    })
                    .map(|s| (s.id, s.position))
                    .collect();
                for (cid, cpos) in candidates {
                    if m.distance_sqr(cpos) > range * range {
                        continue;
                    }
                    if self.beyond_sqr.is_some_and(|d| m.distance_sqr(cpos) <= d) {
                        continue;
                    }
                    if !m.sees(cx.w, Who::Mob(cid)) {
                        continue;
                    }
                    self.found = Some(Who::Mob(cid));
                    break;
                }
            }
        }
        self.found.is_some()
    }

 -- can_continue
{
        target_continue(&mut self.target_goal, self.must_see, m, cx)
    }

### crates/horizon-client/src/mobs/goals.rs :: Breed
 -- can_use
{
        if m.extra.in_love <= 0 {
            return false;
        }
        let species = m.species();
        // Wolf.canMate and Cat.canMate: both tame. The sitting pose sits on the
        // partner side only, and only for a wolf (`wolf.isInSittingPose()`), so
        // a tamed mob ordered to sit still walks off to breed.

        if tamable && !m.extra.tame {
            return false;
        }

        if species == Species::Mule {

            return false;

        let equine = matches!(species, Species::Horse | Species::Donkey) || llama;

        // vehicle or a passenger and at full health (a camel is always tame).
        let camel = species == Species::Camel;

        // and at full health.

                || m.rider.is_some()

                || m.extra.mount.is_some()

                    && same
                    && s.in_love
                    && !s.baby
                    && (!tamable || s.tame)

                    && !(species == Species::Wolf && s.sitting)
                    // AbstractHorse.canParent of the partner.

### crates/horizon-client/src/mobs/undead.rs :: MoveThroughVillage
 -- can_use
{
        if !can_navigate_ground(m) {
            return false;
        }
        // updateVisited: the list keeps its last 16.
        if self.visited.len() > 15 {
            self.visited.remove(0);
        }
        if self.only_at_night && cx.w.daylight {
            return false;
        }
        let w = cx.w;
        let block = |p: Vec3| [p.x.floor() as i32, p.y.floor() as i32, p.z.floor() as i32];
        let section = |p: [i32; 3]| [p[0] >> 4, p[1] >> 4, p[2] >> 4];
        let pos = block(m.position);
        let villages = village_sections(m, w);
        // ServerLevel.isCloseToVillage(pos, 6).
        if sections_to_village(&villages, section(pos)) > 6 {
            return false;
        }
        let taken = others_tickets(m, w);
        let visited = &self.visited;
        let land = m.land_random_pos_scored(w, 15, 7, |p| {
            if sections_to_village(&villages, section(p)) > 1 {
                return f64::NEG_INFINITY;
            }
            // The first unvisited taken point within 10, by its distance.
            occupied_pois_in_range(w, &taken, p, 10)
                .into_iter()
                .find(|q| !visited.contains(q))
                .map_or(f64::NEG_INFINITY, |q| {
                    let d = |i: usize| f64::from(q[i] - pos[i]);
                    -(d(0) * d(0) + d(1) * d(1) + d(2) * d(2))
                })
        });
        let Some(land) = land else { return false };
        let Some(target) =
            occupied_pois_in_range(w, &taken, block(Vec3::new(land[0], land[1], land[2])), 10)
                .into_iter()
                .find(|q| !self.visited.contains(q))
        else {
            return false;
        };
        self.poi = target;
        let doors = (self.can_deal_with_doors)(m);
        m.ai.nav.can_open_doors = doors;
        let mut path = m.create_path_to(w, self.poi, 0);
        m.ai.nav.can_open_doors = true;
        if path.is_none() {
            // No way there: a step of up to 10 blocks toward it instead.
            let to = [
                f64::from(self.poi[0]) + 0.5,
                f64::from(self.poi[1]),
                f64::from(self.poi[2]) + 0.5,
            ];
            let Some(step) =
                m.default_random_pos_towards(w, 10, 7, to, f64::from(std::f32::consts::FRAC_PI_2))
            else {
                return false;
            };
            m.ai.nav.can_open_doors = doors;
            path = m.create_path_to(w, block(Vec3::new(step[0], step[1], step[2])), 0);
            m.ai.nav.can_open_doors = true;
            if path.is_none() {
                return false;
            }
        }
        // A wooden door on the way: only as far as the door.
        let door = path.as_ref().and_then(|p| {
            p.nodes
                .iter()
                .find(|n| wooden_door(w, [n.x, n.y + 1, n.z]))
                .map(|n| [n.x, n.y, n.z])
        });
        if let Some(at) = door {
            path = m.create_path_to(w, at, 0);
        }
        self.path = path;
        self.path.is_some()
    }

 -- can_continue
{
        !m.ai.nav.is_done()
            && !closer_to_center(
                self.poi,
                m.position,
                f64::from(m.width()) + f64::from(self.distance),
            )
    }
